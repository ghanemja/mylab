    properties
        x = 0
        y = 0
    end
    methods
        function obj = Pt(x, y)
            if nargin > 0, obj.x = x; obj.y = y; end
        end

    properties (Constant)
        UnitArea = pi
    end
    properties
        r = 1
    end
    properties (Dependent)
        area
    end
    methods
        function a = get.area(obj)
            a = pi * obj.r^2;

        end
    end

### oop/operator_overloading
@file V2.m
classdef V2
    properties
        a = 0
        b = 0
    end
    methods
        function o = V2(a, b)
            if nargin, o.a = a; o.b = b; end
        end
        function r = plus(p, q)
            r = V2(p.a + q.a, p.b + q.b);
        end
        function r = minus(p, q)
            r = V2(p.a - q.a, p.b - q.b);
        end
        function r = mtimes(p, k)
            if isa(p, 'double'), t = p; p = k; k = t; end
            r = V2(p.a * k, p.b * k);
        end
        function r = eq(p, q)
            r = p.a == q.a && p.b == q.b;
        end
        function r = uminus(p)
            r = V2(-p.a, -p.b);
        end
        function r = lt(p, q)
            r = norm([p.a p.b]) < norm([q.a q.b]);
        end
        function disp(p)
            fprintf('V2(%d,%d)\n', p.a, p.b);
        end
    end
end
@main
p = V2(1, 2) + V2(3, 4); assert(p == V2(4, 6)); q = 2 * p; assert(q.a == 8); assert((p - p) == V2(0, 0)); assert(-p == V2(-4, -6));
assert(V2(1, 0) < V2(5, 5)); s = evalc('disp(p)'); assert(strcmp(strtrim(s), 'V2(4,6)'));

### oop/subsref_subsasgn_overload
@file Wrap.m
classdef Wrap
    properties
        data = [10 20 30]
    end
    methods
        function varargout = subsref(obj, s)
            if strcmp(s(1).type, '()')
                varargout{1} = obj.data(s(1).subs{:}) * 2;
            else
                [varargout{1:nargout}] = builtin('subsref', obj, s);
            end
        end
        function obj = subsasgn(obj, s, v)
            if strcmp(s(1).type, '()')
                obj.data(s(1).subs{:}) = v + 1;
            else
                obj = builtin('subsasgn', obj, s, v);
            end
        end
    end
end
@main
w = Wrap(); assert(w(2) == 40); assert(isequal(w.data, [10 20 30])); w(1) = 5; assert(w.data(1) == 6);
