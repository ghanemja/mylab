### oop/held_out_value_and_handle
@file Stack.m
classdef Stack < handle
    properties (SetAccess = private)
        items = {}
    end
    properties (Dependent)
        Count
    end
    methods
        function push(obj, x)
            obj.items{end+1} = x;
        end
        function x = pop(obj)
            if isempty(obj.items), error('Stack:empty', 'stack is empty'); end
            x = obj.items{end}; obj.items(end) = [];
        end
        function n = get.Count(obj)
            n = numel(obj.items);
        end
    end
end
@file Temp.m
classdef Temp
    properties
        C = 0
    end
    properties (Dependent)
        F
    end
    methods
        function obj = Temp(c)
            if nargin, obj.C = c; end
        end
        function f = get.F(obj)
            f = obj.C * 9 / 5 + 32;
        end
        function obj = set.F(obj, f)
            obj.C = (f - 32) * 5 / 9;
        end
        function obj = warm(obj, d)
            obj.C = obj.C + d;
        end
        function r = plus(a, b)
            r = Temp(a.C + b.C);
        end
        function tf = lt(a, b)
            tf = a.C < b.C;
        end
        function s = char(obj)
            s = sprintf('%.1f C', obj.C);
        end
    end
    methods (Static)
        function t = boiling()
            t = Temp(100);
        end
    end
end
@main
s = Stack(); s.push(1); s.push('two'); assert(s.Count == 2); assert(strcmp(s.pop(), 'two')); assert(s.Count == 1);
t = s; t.push(5); assert(s.Count == 2); ok = false; try, s.items = {}; catch, ok = true; end; assert(ok);
s.pop(); s.pop(); id = ''; try, s.pop(); catch e, id = e.identifier; end; assert(strcmp(id, 'Stack:empty'));
a = Temp(100); assert(a.F == 212); a.F = 32; assert(abs(a.C) < 1e-12); b = a.warm(10); assert(a.C == 0 && b.C == 10);
c = Temp(1) + Temp(2); assert(c.C == 3); assert(Temp(1) < Temp(2)); assert(strcmp(char(Temp(21.55)), '21.6 C') || strcmp(char(Temp(21.55)), '21.5 C'));

    properties
        wheels = 4

    end

        end

    end

    end
end

@file Level.m
classdef Level < uint8
    enumeration
        Low (1)
        Mid (5)
        High (9)
    end
end
@file Sensor.m
classdef Sensor < handle
    properties
        value = 0
    end
    events
        Changed
    end
    methods
        function update(obj, v)
            obj.value = v;
            notify(obj, 'Changed');
        end
    end
end
@file Recorder.m
classdef Recorder < handle
    properties
        seen = []
    end
    methods
        function obj = Recorder(s)
            addlistener(s, 'Changed', @(src, evt) obj.record(src));
        end
        function record(obj, src)
            obj.seen(end+1) = src.value;
        end
    end
end
@main

m = enumeration('Level'); assert(numel(m) == 3); switch l, case Level.Low, r = 'l'; case Level.Mid, r = 'm'; otherwise, r = '?'; end; assert(r == 'm');
assert(Level.High > Level.Low); assert(isa(l, 'Level') && isa(l, 'uint8'));
